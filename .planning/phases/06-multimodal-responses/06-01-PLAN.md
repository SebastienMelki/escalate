   ```typescript
   }
   ```

3. **src/transcription/index.ts** -- Define the provider interface and barrel exports:
   ```typescript
   /** Abstract transcription provider for testability and future provider swaps. */
   export interface TranscriptionProvider {
     transcribe(audio: Buffer, mimeType: string, filename: string): Promise<TranscriptionResult>;
   }

   export type { TranscriptionResult } from './types.js';
   export { WhisperTranscriptionProvider } from './whisper.js';
   ```

4. **src/transcription/whisper.ts** -- Implement the OpenAI Whisper provider:
   - Import `OpenAI` and `toFile` from `'openai'`
   - Import `TranscriptionProvider` from `'./index.js'` and `TranscriptionResult` from `'./types.js'`
   - Create `WhisperTranscriptionProvider` class implementing `TranscriptionProvider`:
     - Constructor takes `apiKey: string`, creates `new OpenAI({ apiKey })`
     - `async transcribe(audio: Buffer, mimeType: string, filename: string): Promise<TranscriptionResult>`:
       - Convert buffer to file: `const file = await toFile(audio, filename, { type: mimeType })`
       - Call Whisper API: `const result = await this.client.audio.transcriptions.create({ file, model: 'whisper-1', response_format: 'verbose_json' })`
       - Return `{ text: result.text, durationMs: result.duration ? result.duration * 1000 : undefined }`
   - IMPORTANT: Use `import OpenAI from 'openai'` (default import) and `import { toFile } from 'openai'` (named import) -- check the actual openai SDK exports. The research shows `import OpenAI, { toFile } from 'openai'` as the pattern.

5. **src/index.ts** -- Add transcription barrel export: `export type { TranscriptionProvider, TranscriptionResult } from './transcription/index.js';` and `export { WhisperTranscriptionProvider } from './transcription/whisper.js';`

6. **tsup.config.ts** -- Add `'openai'` to the `external` array so it is not bundled (same as better-sqlite3 and @slack/bolt).
