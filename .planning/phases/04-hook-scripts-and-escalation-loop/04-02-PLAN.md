**scripts/on-permission-request.ts** (HOOK-01):
```
#!/usr/bin/env node
import { readFileSync } from 'node:fs';
import { readPort, createEscalation, pollForResponse } from './lib/bridge-client.js';

async function main(): Promise<void> {
  const input = JSON.parse(readFileSync('/dev/stdin', 'utf-8'));
  const port = readPort();

  const esc = await createEscalation(port, {
    event_type: 'PermissionRequest',
    request_json: JSON.stringify(input),
    fallback_action: 'deny',
    timeout_seconds: 600,
  });

  const result = await pollForResponse(port, esc.escalation_id);

  if (result.status === 'resolved' && result.responseJson) {
    const response = JSON.parse(result.responseJson);
    if (response.type === 'action' && response.actionId === 'approve') {
      process.stdout.write(JSON.stringify({
        hookSpecificOutput: { hookEventName: 'PermissionRequest', decision: { behavior: 'allow' } },
      }));
      process.exit(0);
    }
  }

  // Default: deny
  process.stdout.write(JSON.stringify({
    hookSpecificOutput: {
      hookEventName: 'PermissionRequest',
      decision: { behavior: 'deny', message: 'Permission denied by user via Escalate' },
    },
  }));
  process.exit(0);
}

main().catch(() => process.exit(0));
```

**scripts/on-stop.ts** (HOOK-03):
Different pattern — checks `stop_hook_active` to prevent infinite loops:
```
const input = JSON.parse(readFileSync('/dev/stdin', 'utf-8'));

// Prevent infinite loop: if stop_hook_active, this is a re-stop. Allow it.
if (input.stop_hook_active) {
  process.exit(0);
}

const port = readPort();
const esc = await createEscalation(port, {
  event_type: 'Stop',
  request_json: JSON.stringify(input),
  fallback_action: 'allow',  // On timeout, let Claude stop
  timeout_seconds: 600,
});

const result = await pollForResponse(port, esc.escalation_id);

if (result.status === 'resolved' && result.responseJson) {
  const response = JSON.parse(result.responseJson);
  // If user wants to continue, block the stop with their reason
  if (response.type === 'action' && response.actionId === 'continue') {
    process.stdout.write(JSON.stringify({ decision: 'block', reason: 'User wants to continue via Escalate' }));
    process.exit(0);
  }
  if (response.type === 'text' && response.text) {
    process.stdout.write(JSON.stringify({ decision: 'block', reason: response.text }));
    process.exit(0);
  }
}

// Default: allow stop
process.exit(0);
```

const input = JSON.parse(readFileSync('/dev/stdin', 'utf-8'));
